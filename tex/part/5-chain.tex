\section{Цепи Маркова}

\begin{commentary}
    \textbf{Простыми словами:} Цепь Маркова - это математическая модель системы, которая переключается между разными состояниями.
    
    Главная фишка здесь в \textbf{отсутствии памяти}: 
    то, где система окажется на следующем шаге, зависит \textbf{только} от того, где она находится прямо сейчас. 
    У каждого состаяния есть свои вероятности перехода в остальные состояния (элемент $p_{ij}$ матрицы $P_{N \times N}$ - вероятность перейти из события $s_i$ к $s_j$).
\end{commentary}


\begin{definition}[Цепь Маркова]
\leavevmode

Рассмотрим последовательность случайных величин $S^{(0)}, S^{(1)}, \dots, S^{(n)}$,\\
принимающих значения из конечного множества состояний $\{s_1, s_2, \dots, s_N\}$.

Начальное распределение: $\vec{p}^{(0)} = (p_0^{(0)}, \dots, p_N^{(0)})$.

Матрица переходных вероятностей $P_{N \times N} = \|p_{ij}\|$ удовлетворяет условиям:
\begin{enumerate}
    \item $0 \le p_{ij} \le 1 $ - т.к. это вероятность
    \item $\sum_{j=1}^{N} p_{ij} = 1$ для всех $i$ - полная группа вероятностей
\end{enumerate}
\end{definition}

\begin{statement}[Марковское свойство]
\leavevmode

Вероятность траектории:
$$ P(S^{(0)} = s_{i_0}, S^{(1)} = s_{i_1}, \dots, S^{(n)} = s_{i_n}) = p_{i_0}^{(0)} \cdot p_{i_0 i_1} \cdot p_{i_1 i_2} \cdots p_{i_{n-1} i_n} $$
для строки $s_{i_0}, s_{i_1}, \dots, s_{i_n}$ длины $n+1$.
\end{statement}

\begin{statement}
Сумма вероятностей всех траекторий равна 1:
$$ \sum_{i_0, \dots, i_n} P(S^{(k)} = s_{i_k} \text{ для } k \in [1, n]) = \sum_{i_0} p_{i_0}^{(0)} \sum_{i_1, \dots, i_n} p_{i_0 i_1} \cdots p_{i_{n-1} i_n} = 1 $$
\end{statement}

\begin{statement}
\leavevmode

Вероятность прохождения траектории в момент времени $t$ через $s_i$:
$$ P(S^{(0)} = s_{i_0}, \dots, S^{(t)} = s_{i}, \dots , S^{(n)} = s_{i_n}) = \sum_{i_0, \dots, i_{t-1}, i_{t+1}, \dots, i_n} p_{i_0}^{(0)} \cdot p_{i_0 i_1} \cdots p_{i_{t-1} i}$$
для строк длины $n+1$, где $n \ge t$.
\end{statement}

\begin{statement}
\leavevmode

Вероятность прохождения траектории в момент времени $r$ через $s_j$, при условии, в момент времени $t$ через $s_i$:
$$ P(S^{(0)} = s_{i_0}, \dots, S^{(t)} = s_{i}, \dots, S^{(r)} = s_{j}, \dots , S^{(n)} = s_{i_n}) = \sum_{i_{t+1}, \dots, i_{r-1}} p_{i i_{t+1}} \cdots p_{i_{r-1} j}$$
для строк длины $n+1$, где $n \ge r$.
\end{statement}

\begin{theorem}[Марковское свойство (условная независимость)]
Для цепи Маркова условные вероятности событий при условии промежуточного состояния распадаются на произведение:
$$ P(AC|B) = P(A|B) \cdot P(C|B) $$
где $A = \{S^{(t)} = s_i\}$, $B = \{S^{(r)} = s_j\}$, $C = \{S^{(v)} = s_k\}$ при $t < r < v$.
\end{theorem}

\begin{proof}
\leavevmode

Рассмотрим событие $B = \{S^{(r)} = s_j\}$.
$$ P(B) = \sum_{i_0, i_1, \dots, i_{r-1}, i_{r+1}, \dots, i_n} P(S^{(0)} = s_{i_0}, \dots, S^{(r)} = s_j, \dots, S^{(n)} = s_{i_n}) $$
$$ = \sum p_{i_0}^{(0)} p_{i_0 i_1} \cdots p_{i_{r-1} j} \cdot p_{j i_{r+1}} \cdots p_{i_{n-1} i_n} $$

Для события $BC = \{S^{(t)} = s_i, S^{(v)} = s_j\}$:
$$ P(C) = \sum p_{i_0}^{(0)} p_{i_0 i_1} \cdots p_{i_{t-1} j} \cdot p_{j i_{t+1}} \cdots p_{i_{v-1} k} \cdot p_{k i_{v+1}} \cdots p_{i_{n-1} i_n} $$

Разделим сумму на две части (до момента $r$ и после):
$$ P(BC) = \underbrace{\sum p_{i_0}^{(0)} p_{i_0 i_1} \cdots p_{i_{r-1} j}}_{P(B)} \cdot \underbrace{\sum p_{j i_{r+1}} \cdots p_{i_{v-1} k}}_{P(C|B)} $$

И сравним с $ABC$:
$$ P(ABC) = \underbrace{\sum p_{i_0}^{(0)} p_{i_0 i_1} \cdots p_{i_{t-1} i}}_{P(A)} \cdot \underbrace{\sum p_{i i_{t+1}} \cdots p_{i_{r-1} j}}_{P(B|A)} \cdot \underbrace{\sum p_{j i_{r+1}} \cdots p_{i_{v-1} k}}_{P(C|AB)} $$

Заметим, что $P(C|AB) = P(C|B)$. Тогда:
$$ P(AC|B) = \frac{P(ABC)}{P(B)} = \frac{P(C|AB)P(AB)}{P(B)} = P(A|B) \cdot P(C|AB) = P(A|B) \cdot P(C|B) $$
\end{proof}

\subsection{Сходимость степеней матрицы переходов}

\begin{definition}[Коэффициенты эргодичности]
Для матрицы переходов $P^{(n)} = \|p_{ij}^{(n)}\|$ определим:
$$ Q_j^{(n)} = \max_{1 \le i \le N} p_{ij}^{(n)}, \quad q_j^{(n)} = \min_{1 \le i \le N} p_{ij}^{(n)} $$
\end{definition}

\begin{lemma}
Последовательность $Q_j^{(n)} - q_j^{(n)}$ монотонно не возрастает и стремится к нулю при $n \to \infty$.
\end{lemma}

\begin{proof}
Покажем, что $Q_j^{(n+1)} - q_j^{(n+1)} \le Q_j^{(n)} - q_j^{(n)}$.

Используя формулу Чепмена-Колмогорова:
$$ p_{ij}^{(n+1)} = \sum_{k=1}^{N} p_{ik}^{(n)} p_{kj} $$

Для любых состояний $i_1, i_2$:
$$ p_{i_1 j}^{(n+1)} - p_{i_2 j}^{(n+1)} = \sum_{k=1}^{N} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) p_{kj} $$

Разобьем сумму на две части: $K_+ = \{k: p_{i_1 k}^{(n)} \ge p_{i_2 k}^{(n)}\}$ и $K_- = \{k: p_{i_1 k}^{(n)} < p_{i_2 k}^{(n)}\}$.

Тогда:
$$ p_{i_1 j}^{(n+1)} - p_{i_2 j}^{(n+1)} = \sum_{k \in K_+} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) p_{kj} + \sum_{k \in K_-} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) p_{kj} $$

Оценим каждую сумму:
$$ \le \sum_{k \in K_+} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) Q_j^{(n)} + \sum_{k \in K_-} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) q_j^{(n)} $$

Так как $\sum_{k \in K_+} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) = -\sum_{k \in K_-} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) = h_{i_1 i_2}$, получаем:
$$ p_{i_1 j}^{(n+1)} - p_{i_2 j}^{(n+1)} \le h_{i_1 i_2} (Q_j^{(n)} - q_j^{(n)}) $$

где $h_{i_1 i_2} = \sum_{k \in K_+} (p_{i_1 k}^{(n)} - p_{i_2 k}^{(n)}) \le 1 - NH < 1$ (при условии положительности всех элементов).

Следовательно:
$$ Q_j^{(n+1)} - q_j^{(n+1)} \le (Q_j^{(n)} - q_j^{(n)}) \cdot h \to 0 \text{ при } n \to \infty $$
\end{proof}

\begin{theorem}[Сходимость к стационарному распределению]
Если все элементы матрицы переходов $P$ строго положительны, то:
\begin{enumerate}
    \item Существует предел $\lim_{n \to \infty} p_{ij}^{(n)} = p_j^*$, не зависящий от $i$
    \item Вектор $\vec{p}^* = (p_1^*, \dots, p_N^*)$ является единственным стационарным распределением: $\vec{p}^* = \vec{p}^* P$
\end{enumerate}
\end{theorem}

\begin{proof}
Из леммы следует, что $Q_j^{(n)} - q_j^{(n)} \to 0$, значит $\lim_{n \to \infty} p_{ij}^{(n)} = p_j^*$ существует и не зависит от $i$.

Переходя к пределу в уравнении $p_{ij}^{(n+1)} = \sum_{k} p_{ik}^{(n)} p_{kj}$:
$$ p_j^* = \sum_{k=1}^{N} p_k^* p_{kj} $$

Что доказывает стационарность: $\vec{p}^* = \vec{p}^* P$.

Для единственности: если $\vec{\pi} = \vec{\pi} P$, то $\vec{\pi} = \vec{\pi} P^n$ для всех $n$. Переходя к пределу:
$$ \pi_j = \sum_{k=1}^{N} \pi_k p_{kj}^{(n)} \xrightarrow{n \to \infty} \sum_{k=1}^{N} \pi_k p_j^* = p_j^* \sum_{k=1}^{N} \pi_k = p_j^* $$
\end{proof}

\begin{note}[Независимость от начального состояния]
Для любой начальной распределения $\vec{p}^{(0)}$:
$$ \vec{p}^{(n)} = \vec{p}^{(0)} P^n \xrightarrow{n \to \infty} \vec{p}^* $$
\end{note}
